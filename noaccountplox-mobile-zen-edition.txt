! Skip onelink.me tracking redirects -----------------------------------------------
! REQUIRE: Allow custom filters requiring trust [Settings → Filter lists → Enable "Allow custom filters requiring trust"]
! Bypasses TikTok mobile web redirect/interstitial behavior by skipping OneLink deep-link tracking URLs and opening the canonical TikTok URL directly. This can help prevent forced sign-up/app-open prompts and keep browsing the “For You” feed.
||snssdk1233.onelink.me/bIdt*$urlskip=?canonical -uricomponent,important
||onelink.me^$urlskip=?canonical -uricomponent,important
! Fallback para enlaces con af_dp
||onelink.me^$urlskip=?af_dp -uricomponent -uricomponent,important,domain=tiktok.com
! END-REQUIRE

! Hide TikTok entire right-side action column
! Includes:
! - Avatar/follow button
! - Like button
! - Comment button
! - Share button
! - Music/spinning album button
tiktok.com##div[class*="DivRightAreaBox"]

! Hide TikTok video caption/description overlay
! Includes:
! - Username
! - Caption text
! - Hashtags
! - "more" expand button
! - Music title section
!tiktok.com##div[class*="DivVideoCaption"]
! Disable clicks/interactions on TikTok video caption overlay
tiktok.com##div[class*="DivVideoCaption"]:style(pointer-events: none !important;)

! Hide TikTok top header bar
! Includes:
! - Hamburger/menu button
! - "Open App" button
! - Search button
tiktok.com##div[class*="DivHeaderContainer"]
! Hide TikTok hamburger/menu button in the top navigation bar
! tiktok.com##div[data-e2e="view-nav-arrow"]
! Hide TikTok header hamburger container (Backup)
! tiktok.com##div[class*="DivHamburgerContainer"]
! Hide TikTok "Open App" button in the top header
tiktok.com##button[class*="ButtonCTAOpenApp"]

! Bottom bar -----------------------------------------------
! Remove TikTok discover tab container
tiktok.com##div[class*="DivTabContainer"]:has(a[data-e2e="discover-icon"])
! Hide TikTok Upload button
tiktok.com##div[class*="matrix-smart-wrapper"]:has(div[data-e2e="open-titok-icon"])
! Hide TikTok inbox/message tab button in bottom navigation
tiktok.com##div[class*="DivTabContainer"]:has(a[data-e2e="dm-icon"])
! Remove TikTok profile tab container
tiktok.com##div[class*="DivTabContainer"]:has(a[data-e2e="profile-icon"])

! Discover Tab -----------------------------------------------
! Hide TikTok floating "Discover more" button (just as a backup, this page is blocked above)
tiktok.com##div[class*="DivFloatButtonWrapper"]
